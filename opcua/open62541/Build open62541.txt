подключением к проекту!!!

# 4. Конфигурация для MinGW
# Сборка только статической библиотеки (без шифрования)

  -DCMAKE_INSTALL_PREFIX="C:\Qt\Tools\mingw1310_64\bin" `
  -DCMAKE_PREFIX_PATH="C:\Qt\Tools\mingw1310_64\bin" `
  -DUA_ENABLE_AMALGAMATION=ON `
  -DCMAKE_BUILD_TYPE=Release `
  -DUA_ENABLE_ENCRYPTION_OPENSSL=ON `
  -DUA_ENABLE_ENCRYPTION_MBEDTLS=ON `
  -DBUILD_SHARED_LIBS=OFF ..

# Сборка только статической библиотеки (с поддержкой шифрования)
cmake .. -G "MinGW Makefiles" `
	-DCMAKE_INSTALL_PREFIX="C:\Qt\Tools\mingw1310_64\bin" `
	-DCMAKE_PREFIX_PATH="C:\Qt\Tools\mingw1310_64\bin" `
	-DOPENSSL_ROOT_DIR="C:\Qt\Tools\OpenSSLv3\Win_x64" `
	-DUA_ENABLE_AMALGAMATION=ON `
	-DUA_AMALGAMATION_MULTIARCH=ON `
	-DUA_MULTITHREADING=0 `
	-DCMAKE_BUILD_TYPE=Release `
	-DUA_ENABLE_ENCRYPTION=OPENSSL ..

# Сборка динамической библиотеки (-DBUILD_SHARED_LIBS=ON - главный флаг для сборки shared libraries)
# (не собирается т.к. для компилятора необходимо установить libssp.
